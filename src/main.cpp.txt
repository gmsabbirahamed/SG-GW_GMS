/*
===     network     ===
network task workflow-
set mqtt
boot the modem (hard reset/off-on and test AT) ->if failed esp restart.
initialized modem (waitfornetwork and ..) -> if failed 3 times esp restart
connect to network (dis connect gprs & connect gprs with apn,user,pass)
if gprs is connected & mqtt is not connected -> try to connect/reconnect mqtt
if mqtt is connected && gprs is connected = online
if mqtt is not connected && gprs is not connected = offline
[if failed to connect gprs for 3 times that means gsm error -> wait for 10sec to retry gprs connection.
    esp restart after 3 gsm error]

===     main     ===
main task workflow-
send heartbeat in 1min interval
if device is online and data is availabe in queue -> send data to mqtt.
can be detect single press, double press, 3sec, 5sec, 10sec press.
[show real time led color status like during press and hold the button,, if complete 3sec it show solid green, then when complete 5sec the color change to solid blue, then when complete hold 10sec or more color change to solid red. but message will print after release the button..]

===      led status      ===
connecting network -> red blinking
offline -> solid red
online -> black
mqtt message received -> single blue blink
rf signal detect -> single white blink
heartbeat send -> single cyan blink
data send -> single green blink

===     rf sensor task      ===
read rf sensor and enqueue in queue - max 50 (if queue is full dequeue oldest data and enqueue new)

===     button task      ===
detect button press properly 
    ->show color status
    ->print message
    ->do action (updaet later)


===     special     ===
ota task will create dynamically ->when get command from mqtt msg, suspend/delete all other task & create ota task.
*/

// ==================== Task Implementations ====================
void mainTask(void* parameter) {
    unsigned long lastHeartbeatTime = 0;
    RFData rfData;
    MQTTMessage mqttMsg;
    ButtonEvent btnEvent;
    
    SerialMon.println("Main Task started");
    
    while (1) {
        if (otaInProgress) {
            vTaskDelay(pdMS_TO_TICKS(1000));
            continue;
        }
        
        // Send heartbeat every minute
        if (millis() - lastHeartbeatTime >= HEARTBEAT_INTERVAL) {
            if (deviceOnline) {
                publishHeartbeat();
                sendLedCommand(LED_HEARTBEAT);
                lastHeartbeatTime = millis();
            }
        }
        
        // Check for RF data in queue and publish if online
        if (deviceOnline && uxQueueMessagesWaiting(rfDataQueue) > 0) {
            if (xQueueReceive(rfDataQueue, &rfData, 0) == pdTRUE) {
                // Create MQTT message
                snprintf(mqttMsg.topic, sizeof(mqttMsg.topic), "%s", pubTopic);
                // snprintf(mqttMsg.payload, sizeof(mqttMsg.payload), 
                //          "{\"device\":\"%s\",\"type\":\"rf\",\"value\":%lu,\"timestamp\":%lu}",
                //          DEVICE_ID.c_str(), rfData.rf_value, rfData.rf_length);
                String payload = DEVICE_ID + "," + rfData.rf_value + "," + rfData.rf_length ;
                Serial.println(payload);
                // mqttMsg.payload = payload.c_str();
                snprintf(mqttMsg.payload, sizeof(mqttMsg.payload), "%s", payload.c_str());
                // mqtt.publish(mqttMsg.topic, payload.c_str());

                // Send to MQTT publish queue
                if (xQueueSend(mqttPublishQueue, &mqttMsg, pdMS_TO_TICKS(100)) == pdTRUE) {
                    sendLedCommand(LED_PUBLISH_RF);
                    SerialMon.printf("MainTask: RF data queued - Value: %lu\n", rfData.rf_value);
                }
            }
        }
        
        
        vTaskDelay(pdMS_TO_TICKS(100));
    }
}

// ==================== Helper Functions ====================

String mqttStateToText(int state) {
    switch (state) {
        case MQTT_CONNECTION_TIMEOUT: return "Timeout";
        case MQTT_CONNECTION_LOST: return "Lost";
        case MQTT_CONNECT_FAILED: return "Failed";
        case MQTT_DISCONNECTED: return "Disconnected";
        case MQTT_CONNECT_BAD_PROTOCOL: return "Bad Protocol";
        case MQTT_CONNECT_BAD_CLIENT_ID: return "Bad Client ID";
        case MQTT_CONNECT_UNAVAILABLE: return "Unavailable";
        case MQTT_CONNECT_BAD_CREDENTIALS: return "Bad Credentials";
        case MQTT_CONNECT_UNAUTHORIZED: return "Unauthorized";
        default: return "Unknown";
    }
}

bool performHardModemReset() {
    SerialMon.println("Network: Performing HARD modem reset (power cycle)");
    networkStats.modem_power_cycles++;
    
    // Hardware power cycle
    digitalWrite(MODEM_PWR, LOW);
    vTaskDelay(pdMS_TO_TICKS(1000));
    digitalWrite(MODEM_PWR, HIGH);
    vTaskDelay(pdMS_TO_TICKS(8000));  // Extended boot time
    
    // Verify with AT commands
    for (int i = 0; i < 5; i++) {
        SerialAT.println("AT");
        vTaskDelay(pdMS_TO_TICKS(200));
        
        if (SerialAT.find("OK")) {
            SerialMon.println("Network: Modem responded after power cycle");
            return true;
        }
        vTaskDelay(pdMS_TO_TICKS(500));
    }
    
    strcpy(networkStats.last_error, "Modem power cycle failed");
    return false;
}

bool initializeModemWithRetry() {
    SerialMon.println("Network: Initializing modem...");
    sendLedCommand(LED_GSM_INIT);
    
    if (xSemaphoreTake(modemMutex, pdMS_TO_TICKS(5000)) == pdTRUE) {
        // Try modem restart
        if (!modem.restart()) {
            SerialMon.println("Network: Modem restart failed");
            xSemaphoreGive(modemMutex);
            networkStats.modem_init_failures++;
            return false;
        }
        
        // Get modem info
        String modemInfo = modem.getModemInfo();
        SerialMon.print("Network: Modem Info: ");
        SerialMon.println(modemInfo);
        
        // Check SIM status
        if (modem.getSimStatus() != SIM_READY) {
            SerialMon.println("Network: SIM not ready");
            xSemaphoreGive(modemMutex);
            networkStats.modem_init_failures++;
            strcpy(networkStats.last_error, "SIM not ready");
            return false;
        }
        
        xSemaphoreGive(modemMutex);
        return true;
    }
    
    return false;
}

bool waitForNetworkRegistration() {
    SerialMon.println("Network: Waiting for network registration...");
    
    if (xSemaphoreTake(modemMutex, pdMS_TO_TICKS(5000)) == pdTRUE) {
        // Check signal quality first
        int signalQuality = modem.getSignalQuality();
        networkStats.last_signal_quality = signalQuality;
        SerialMon.printf("Network: Signal quality: %d\n", signalQuality);
        
        // Wait for network with timeout
        if (modem.waitForNetwork(netConfig.network_reg_timeout)) {
            SerialMon.println("Network: Registered to network");
            xSemaphoreGive(modemMutex);
            return true;
        } else {
            SerialMon.println("Network: Network registration timeout");
            xSemaphoreGive(modemMutex);
            strcpy(networkStats.last_error, "Network registration timeout");
            return false;
        }
    }
    
    return false;
}

bool connectGPRS() {
    SerialMon.println("Network: Connecting to GPRS...");
    sendLedCommand(LED_CONNECTING);
    
    if (xSemaphoreTake(modemMutex, pdMS_TO_TICKS(5000)) == pdTRUE) {
        // Disconnect any existing connection
        if (modem.isGprsConnected()) {
            modem.gprsDisconnect();
            vTaskDelay(pdMS_TO_TICKS(1000));
        }
        
        // Connect with timeout
        unsigned long gprsStart = millis();
        bool connected = false;
        
        while (!connected && (millis() - gprsStart < netConfig.gprs_timeout)) {
            connected = modem.gprsConnect(netConfig.apn, netConfig.user, netConfig.pass);
            
            if (!connected) {
                vTaskDelay(pdMS_TO_TICKS(5000));
            }
        }
        
        if (connected) {
            String ip = modem.getLocalIP();
            SerialMon.printf("Network: GPRS connected, IP: %s\n", ip.c_str());
            
            // Verify internet connectivity
            bool internetOk = false;
            if (netClient.connect("8.8.8.8", 53)) {  // DNS test
                internetOk = true;
                netClient.stop();
            }
            
            if (internetOk) {
                SerialMon.println("Network: Internet connectivity verified");
                xSemaphoreGive(modemMutex);
                return true;
            } else {
                SerialMon.println("Network: No internet connectivity");
                modem.gprsDisconnect();
                xSemaphoreGive(modemMutex);
                strcpy(networkStats.last_error, "No internet connectivity");
                return false;
            }
        }
        
        xSemaphoreGive(modemMutex);
        strcpy(networkStats.last_error, "GPRS connection failed");
        return false;
    }
    
    return false;
}

bool connectMQTT() {
    SerialMon.println("Network: Connecting to MQTT...");
    sendLedCommand(LED_MQTT_CONNECTING);
    
    if (xSemaphoreTake(mqttMutex, pdMS_TO_TICKS(5000)) == pdTRUE) {
        String clientId = "GMS-" + DEVICE_ID + "-" + String(esp_random(), HEX);
        
        // Set last will
        String willTopic = String(pubTopic) + "/status";
        String willMessage = "{\"device\":\"" + DEVICE_ID + "\",\"status\":\"offline\"}";
        
        bool connected = mqtt.connect(clientId.c_str(), 
                                      mqttUser, 
                                      mqttPass,
                                      willTopic.c_str(),
                                      0,  // QoS
                                      1,  // retain
                                      willMessage.c_str());
        
        if (connected) {
            SerialMon.println("Network: MQTT connected");
            
            if (mqtt.subscribe(subTopic)) {
                SerialMon.printf("Network: Subscribed to %s\n", subTopic);
            }
            
            // Publish online status
            mqtt.publish(pubTopic, ("{\"device\":\"" + DEVICE_ID + "\",\"status\":\"online\"}").c_str());
            
            xSemaphoreGive(mqttMutex);
            return true;
        } else {
            SerialMon.printf("Network: MQTT failed, rc=%d (%s)\n", 
                           mqtt.state(), mqttStateToText(mqtt.state()));
            xSemaphoreGive(mqttMutex);
            strcpy(networkStats.last_error, "MQTT connection failed");
            return false;
        }
    }
    
    return false;
}

void disconnectAll() {
    SerialMon.println("Network: Disconnecting all connections");
    
    if (xSemaphoreTake(mqttMutex, pdMS_TO_TICKS(1000)) == pdTRUE) {
        mqtt.disconnect();
        xSemaphoreGive(mqttMutex);
    }
    
    if (xSemaphoreTake(modemMutex, pdMS_TO_TICKS(1000)) == pdTRUE) {
        if (modem.isGprsConnected()) {
            modem.gprsDisconnect();
        }
        xSemaphoreGive(modemMutex);
    }
    
    mqttConnected = false;
    gprsConnected = false;
    deviceOnline = false;
}

// ==================== State Handlers ====================

void handleOfflineState() {
    SerialMon.println("Network: OFFLINE state - starting connection sequence");
    sendLedCommand(LED_OFFLINE);
    
    networkStats.connection_attempts++;
    currentNetworkState = NETWORK_STATE_MODEM_POWER_CYCLE;
    stateEntryTime = millis();
}

void handleModemPowerCycle() {
    SerialMon.println("Network: Power cycling modem...");
    
    // Perform two power cycles as per requirement
    for (int cycle = 1; cycle <= 2; cycle++) {
        SerialMon.printf("Network: Power cycle %d/2\n", cycle);
        
        if (!performHardModemReset()) {
            SerialMon.println("Network: Power cycle failed - restarting ESP");
            vTaskDelay(pdMS_TO_TICKS(2000));
            ESP.restart();
        }
        
        if (cycle < 2) {
            vTaskDelay(pdMS_TO_TICKS(5000));  // Wait between cycles
        }
    }
    
    currentNetworkState = NETWORK_STATE_MODEM_INIT;
    networkStats.modem_init_retry_count = 0;
    stateEntryTime = millis();
}

void handleModemInit() {
    SerialMon.println("Network: Initializing modem...");
    
    if (initializeModemWithRetry()) {
        currentNetworkState = NETWORK_STATE_WAIT_FOR_NETWORK;
        networkStats.modem_init_retry_count = 0;
    } else {
        networkStats.modem_init_retry_count++;
        
        if (networkStats.modem_init_retry_count >= netConfig.max_modem_init_retries) {
            SerialMon.println("Network: Modem init failed after 3 retries - restarting ESP");
            vTaskDelay(pdMS_TO_TICKS(2000));
            ESP.restart();
        }
        
        SerialMon.printf("Network: Modem init failed (%d/3), retrying...\n", 
                        networkStats.modem_init_retry_count);
        vTaskDelay(pdMS_TO_TICKS(5000));
    }
    stateEntryTime = millis();
}

void handleWaitForNetwork() {
    SerialMon.println("Network: Waiting for network registration...");
    
    if (waitForNetworkRegistration()) {
        currentNetworkState = NETWORK_STATE_GPRS_CONNECT;
        networkStats.gprs_retry_count = 0;
    } else {
        networkStats.modem_init_failures++;
        currentNetworkState = NETWORK_STATE_RECOVERY_WAIT;
    }
    stateEntryTime = millis();
}

void handleGprsConnect() {
    SerialMon.println("Network: Connecting GPRS...");
    
    if (connectGPRS()) {
        currentNetworkState = NETWORK_STATE_GPRS_CONNECTED;
        gprsConnected = true;
        networkStats.gprs_retry_count = 0;
        networkStats.gprs_failures = 0;
    } else {
        networkStats.gprs_retry_count++;
        networkStats.gprs_failures++;
        
        if (networkStats.gprs_retry_count >= netConfig.max_gprs_retries) {
            SerialMon.println("Network: GPRS failed after 3 retries - going to recovery");
            currentNetworkState = NETWORK_STATE_RECOVERY_WAIT;
        } else {
            SerialMon.printf("Network: GPRS attempt %d/%d failed, retrying...\n", 
                           networkStats.gprs_retry_count, netConfig.max_gprs_retries);
            vTaskDelay(pdMS_TO_TICKS(5000));
        }
    }
    stateEntryTime = millis();
}

void handleGprsConnected() {
    // GPRS is connected, now connect MQTT
    currentNetworkState = NETWORK_STATE_MQTT_CONNECT;
    mqttReconnectAttempts = 0;
    mqttReconnecting = false;
    stateEntryTime = millis();
}

void handleMqttConnect() {
    SerialMon.println("Network: Connecting MQTT...");
    
    if (connectMQTT()) {
        mqttConnected = true;
        deviceOnline = true;
        currentNetworkState = NETWORK_STATE_ONLINE;
        networkStats.successful_connections++;
        networkStats.consecutive_failures = 0;
        networkStats.mqtt_failures = 0;
        sendLedCommand(LED_ONLINE);
        SerialMon.println("Network: ONLINE - All connections established");
    } else {
        networkStats.mqtt_failures++;
        networkStats.mqtt_retry_count++;
        
        if (networkStats.mqtt_retry_count >= netConfig.max_mqtt_retries) {
            SerialMon.println("Network: MQTT failed after 5 retries - going to recovery");
            currentNetworkState = NETWORK_STATE_RECOVERY_WAIT;
        } else {
            SerialMon.printf("Network: MQTT attempt %d/%d failed, retrying...\n", 
                           networkStats.mqtt_retry_count, netConfig.max_mqtt_retries);
            vTaskDelay(pdMS_TO_TICKS(3000));
        }
    }
    stateEntryTime = millis();
}

void processMqttPublishQueue() {
    if (uxQueueMessagesWaiting(mqttPublishQueue) == 0) return;
    
    MQTTMessage msg;
    int processedCount = 0;
    const int MAX_PROCESS_PER_CYCLE = 5;
    
    while (processedCount < MAX_PROCESS_PER_CYCLE && 
           xQueueReceive(mqttPublishQueue, &msg, 0) == pdTRUE) {
        
        if (xSemaphoreTake(mqttMutex, pdMS_TO_TICKS(500)) == pdTRUE) {
            bool published = mqtt.publish(msg.topic, msg.payload);
            
            if (published) {
                SerialMon.printf("Network: Published - %s\n", msg.payload);
                processedCount++;
            } else {
                // Re-queue at front for retry
                SerialMon.println("Network: Publish failed, will retry");
                xQueueSendToFront(mqttPublishQueue, &msg, 0);
                xSemaphoreGive(mqttMutex);
                break;  // Stop processing to allow MQTT to recover
            }
            
            xSemaphoreGive(mqttMutex);
        } else {
            // Couldn't get mutex, put message back
            xQueueSendToFront(mqttPublishQueue, &msg, 0);
            break;
        }
        
        // Small delay between publishes
        vTaskDelay(pdMS_TO_TICKS(10));
    }
}


void handleOnlineState() {
    static unsigned long lastConnectionCheck = 0;
    
    // Monitor connection status
    if (millis() - lastConnectionCheck > 30000) {  // Every 30 seconds
        lastConnectionCheck = millis();
        
        // Check GPRS connection
        bool gprsOk = false;
        if (xSemaphoreTake(modemMutex, pdMS_TO_TICKS(1000)) == pdTRUE) {
            gprsOk = modem.isGprsConnected();
            xSemaphoreGive(modemMutex);
        }
        
        // Check MQTT connection
        bool mqttOk = false;
        if (xSemaphoreTake(mqttMutex, pdMS_TO_TICKS(1000)) == pdTRUE) {
            mqttOk = mqtt.connected();
            xSemaphoreGive(mqttMutex);
        }
        
        // Update status flags
        gprsConnected = gprsOk;
        mqttConnected = mqttOk;
        deviceOnline = (gprsOk && mqttOk);
        
        // Handle offline condition
        if (!deviceOnline) {
            SerialMon.println("Network: Device went offline");
            
            if (gprsOk && !mqttOk) {
                // Case: GPRS connected but MQTT not connected
                SerialMon.println("Network: GPRS OK, MQTT down - attempting MQTT reconnect");
                mqttReconnecting = true;
                mqttReconnectAttempts = 0;
                currentNetworkState = NETWORK_STATE_MQTT_CONNECT;
            } 
            else if (!gprsOk) {
                // Case: GPRS disconnected
                SerialMon.println("Network: GPRS lost - full reconnection needed");
                disconnectAll();
                currentNetworkState = NETWORK_STATE_RECOVERY_WAIT;
            }
            
            stateEntryTime = millis();
            return;
        }
        
        // Update signal quality
        if (xSemaphoreTake(modemMutex, pdMS_TO_TICKS(1000)) == pdTRUE) {
            networkStats.last_signal_quality = modem.getSignalQuality();
            xSemaphoreGive(modemMutex);
        }
    }
    
    // Normal MQTT loop
    if (xSemaphoreTake(mqttMutex, pdMS_TO_TICKS(100)) == pdTRUE) {
        mqtt.loop();
        xSemaphoreGive(mqttMutex);
    }
    
    // Process publish queue
    processMqttPublishQueue();
}

void handleRecoveryWait() {
    if (recoveryWaitStart == 0) {
        recoveryWaitStart = millis();
        networkStats.consecutive_failures++;
        networkStats.failed_connections++;
        networkStats.recovery_wait_count++;
        
        SerialMon.printf("Network: RECOVERY WAIT - failure #%lu, waiting %lu ms\n", 
                        networkStats.consecutive_failures, netConfig.recovery_wait_time);
        sendLedCommand(LED_OFFLINE);
        
        // Clean up connections
        disconnectAll();
    }
    
    // Check if wait time is complete
    if (millis() - recoveryWaitStart >= netConfig.recovery_wait_time) {
        recoveryWaitStart = 0;
        
        // Check if we've exceeded max consecutive failures
        if (networkStats.consecutive_failures >= netConfig.max_consecutive_failures) {
            SerialMon.println("Network: Too many consecutive failures - deep sleep recovery");
            currentNetworkState = NETWORK_STATE_DEEP_SLEEP_RECOVERY;
        } else {
            // Start fresh connection attempt
            currentNetworkState = NETWORK_STATE_MODEM_POWER_CYCLE;
        }
        stateEntryTime = millis();
    }
}

void handleDeepSleepRecovery() {
    SerialMon.println("Network: Deep sleep recovery mode");
    
    // Save stats
    preferences.begin("network", false);
    preferences.putUInt("failures", networkStats.consecutive_failures);
    preferences.putUInt("last_attempt", millis());
    preferences.end();
    
    // Enter deep sleep
    SerialMon.printf("Network: Entering deep sleep for %lu minutes\n", 
                    netConfig.deep_sleep_recovery_time / 60000);
    delay(100);
    
    esp_sleep_enable_timer_wakeup(netConfig.deep_sleep_recovery_time * 1000ULL);
    esp_deep_sleep_start();
}







void networkTask(void* parameter) {
    SerialMon.println("Network Task started with enhanced state machine");
    sendLedCommand(LED_OFFLINE);
    
    // Configure MQTT
    mqtt.setServer(netConfig.broker, netConfig.port);
    mqtt.setKeepAlive(60);
    mqtt.setSocketTimeout(30);
    mqtt.setCallback(mqttCallback);
    mqtt.setBufferSize(1024);
    
    unsigned long lastStatsLog = 0;
    
    while (1) {
        if (otaInProgress) {
            vTaskDelay(pdMS_TO_TICKS(1000));
            continue;
        }
        
        // Log stats every 5 minutes
        if (millis() - lastStatsLog > 300000) {
            logNetworkStats();
            lastStatsLog = millis();
        }
        
        // State machine
        switch (currentNetworkState) {
            case NETWORK_STATE_OFFLINE:
                handleOfflineState();
                break;
                
            case NETWORK_STATE_MODEM_POWER_CYCLE:
                handleModemPowerCycle();
                break;
                
            case NETWORK_STATE_MODEM_INIT:
                handleModemInit();
                break;
                
            case NETWORK_STATE_WAIT_FOR_NETWORK:
                handleWaitForNetwork();
                break;
                
            case NETWORK_STATE_GPRS_CONNECT:
                handleGprsConnect();
                break;
                
            case NETWORK_STATE_GPRS_CONNECTED:
                handleGprsConnected();
                break;
                
            case NETWORK_STATE_MQTT_CONNECT:
                handleMqttConnect();
                break;
                
            case NETWORK_STATE_ONLINE:
                handleOnlineState();
                break;
                
            case NETWORK_STATE_RECOVERY_WAIT:
                handleRecoveryWait();
                break;
                
            case NETWORK_STATE_DEEP_SLEEP_RECOVERY:
                handleDeepSleepRecovery();
                break;
        }
        
        vTaskDelay(pdMS_TO_TICKS(100));
    }
}


void buttonTask(void* parameter) {
    // Initialize button with Bounce2
    button.attach(BUTTON_PIN, INPUT_PULLUP);
    button.interval(BUTTON_DEBOUNCE_INTERVAL);
    button.setPressedState(LOW);  // Button is active low
    
    unsigned long pressStartTime = 0;
    bool isPressing = false;
    int currentHoldThreshold = 0;  // 0=none, 3=3sec, 5=5sec, 10=10sec
    int lastPrintedThreshold = 0;  // To avoid repeated messages
    
    SerialMon.println("Button Task started");
    
    while (1) {
        if (otaInProgress) {
            vTaskDelay(pdMS_TO_TICKS(1000));
            continue;
        }
        
        // Handle ISR flag
        if (buttonInterruptFlag) {
            portENTER_CRITICAL(&buttonMux);
            buttonInterruptFlag = false;
            portEXIT_CRITICAL(&buttonMux);
            button.update();
        } else {
            button.update();
        }
        
        unsigned long currentTime = millis();
        
        // ========== BUTTON PRESS DETECTION ==========
        if (button.pressed()) {
            // Button just pressed
            pressStartTime = currentTime;
            isPressing = true;
            currentHoldThreshold = 0;
            lastPrintedThreshold = 0;
            
            SerialMon.println("Button pressed - showing yellow blink");
            sendLedCommand(LED_BUTTON_PRESS);  // Yellow flash
        }
        
        // ========== VISUAL FEEDBACK DURING HOLD ==========
        if (isPressing) {
            unsigned long holdDuration = currentTime - pressStartTime;
            
            // Determine current threshold based on hold duration
            int newThreshold = 0;
            if (holdDuration >= 10000) {
                newThreshold = 10;
            } else if (holdDuration >= 5000) {
                newThreshold = 5;
            } else if (holdDuration >= 3000) {
                newThreshold = 3;
            }
            
            // Update LED and print message ONLY when threshold changes
            if (newThreshold != currentHoldThreshold) {
                currentHoldThreshold = newThreshold;
                
                switch (currentHoldThreshold) {
                    case 3:
                        sendLedCommand(LED_BUTTON_HOLD_3SEC);  // Solid cyan
                        if (lastPrintedThreshold != 3) {
                            SerialMon.println("Hold: 3 seconds reached - LED cyan");
                            lastPrintedThreshold = 3;
                        }
                        break;
                    case 5:
                        sendLedCommand(LED_BUTTON_HOLD_5SEC);  // Solid pink
                        if (lastPrintedThreshold != 5) {
                            SerialMon.println("Hold: 5 seconds reached - LED pink");
                            lastPrintedThreshold = 5;
                        }
                        break;
                    case 10:
                        sendLedCommand(LED_BUTTON_HOLD_10SEC);  // Solid purple
                        if (lastPrintedThreshold != 10) {
                            SerialMon.println("Hold: 10 seconds reached - LED purple");
                            lastPrintedThreshold = 10;
                        }
                        break;
                }
            }
        }
        
        // ========== BUTTON RELEASE DETECTION ==========
        if (button.released() && isPressing) {
            unsigned long holdDuration = currentTime - pressStartTime;
            isPressing = false;
            
            // Print appropriate message based on hold duration
            SerialMon.println("\n========== BUTTON EVENT ==========");
            
            if (holdDuration >= 10000) {
                SerialMon.println("10 SECOND HOLD DETECTED");
                SerialMon.printf("Hold duration: %lu ms\n", holdDuration);
                SerialMon.println("Final LED was: Purple");
            }
            else if (holdDuration >= 5000) {
                SerialMon.println("5 SECOND HOLD DETECTED");
                SerialMon.printf("Hold duration: %lu ms\n", holdDuration);
                SerialMon.println("Final LED was: Pink");
            }
            else if (holdDuration >= 3000) {
                SerialMon.println("3 SECOND HOLD DETECTED");
                SerialMon.printf("Hold duration: %lu ms\n", holdDuration);
                SerialMon.println("Final LED was: Cyan");
            }
            else {
                SerialMon.println("SINGLE CLICK DETECTED");
                SerialMon.printf("Hold duration: %lu ms\n", holdDuration);
                SerialMon.println("LED: Yellow blink");
            }
            
            SerialMon.println("===================================\n");
            
            // Return LED to network state
            vTaskDelay(pdMS_TO_TICKS(300));  // Short delay to show the final color
            
            if (deviceOnline) {
                sendLedCommand(LED_ONLINE);  // Black (off)
                SerialMon.println("LED: Returning to ONLINE state (OFF)");
            } else if (gprsConnected) {
                sendLedCommand(LED_CONNECTING);  // Red blinking
                SerialMon.println("LED: Returning to CONNECTING state (red blink)");
            } else {
                sendLedCommand(LED_OFFLINE);  // Solid red
                SerialMon.println("LED: Returning to OFFLINE state (solid red)");
            }
        }
        
        vTaskDelay(pdMS_TO_TICKS(50));  // 50ms check interval
    }
}

void rfSensorTask(void* parameter) {
    rfSwitch.enableReceive(RF_SENSOR_PIN);
    
    SerialMon.println("RF Sensor Task started");
    
    uint32_t lastRFValue = 0;
    unsigned long lastRFTime = 0;
    const unsigned long DEBOUNCE_TIME = 800;
    
    while (1) {
        if (otaInProgress) {
            vTaskDelay(pdMS_TO_TICKS(1000));
            continue;
        }
        
        if (rfSwitch.available()) {
            uint32_t rfValue = rfSwitch.getReceivedValue();
            uint32_t rfLength = rfSwitch.getReceivedBitlength();
            rfSwitch.resetAvailable();
            
            // Debounce: Ignore same value within DEBOUNCE_TIME
            if (rfValue == lastRFValue && millis() - lastRFTime < DEBOUNCE_TIME) {
                vTaskDelay(pdMS_TO_TICKS(10));
                continue;
            }
            
            lastRFValue = rfValue;
            lastRFTime = millis();
            
            // Create RF data
            RFData rfData;
            rfData.timestamp = millis();
            rfData.rf_value = rfValue;
            rfData.rf_length = rfLength;
            rfData.sensor_id = 1;
            
            // Check if queue is full
            if (uxQueueMessagesWaiting(rfDataQueue) >= RF_DATA_QUEUE_SIZE) {
                // Remove oldest item
                RFData oldData;
                xQueueReceive(rfDataQueue, &oldData, 0);
                SerialMon.println("RFSensorTask: Queue full, removed oldest item");
            }
            
            // Add to queue
            if (xQueueSend(rfDataQueue, &rfData, 0) == pdTRUE) {
                sendLedCommand(LED_RF_DETECT);
                SerialMon.printf("RFSensorTask: RF detected - Value: %lu\n", rfValue);
            }
        }
        
        vTaskDelay(pdMS_TO_TICKS(10));
    }
}

// ==================== OTA Task ====================
void otaTask(void* parameter) {
    SerialMon.println("OTA Task started");
    sendLedCommand(LED_OTA_IN_PROGRESS);

    suspendAllTasks();

    bool otaOk = false;

    if (checkForNewFirmware()) {
        if (xSemaphoreTake(modemMutex, portMAX_DELAY)) {
            otaOk = performOTAUpdate();
            xSemaphoreGive(modemMutex);
        }
    }

    if (otaOk) {
        Serial.println("[OTA] Success, rebooting");
        delay(1500);
        ESP.restart();
    }

    // OTA failed → clean GSM state
    modem.gprsDisconnect();
    vTaskDelay(pdMS_TO_TICKS(2000));

    resumeAllTasks();

    otaTaskHandle = NULL;
    vTaskDelete(NULL);
}

bool powerCycleModem() {
    SerialMon.println("NetworkTask: Enhanced power cycling modem...");
    
    // Method 1: Hardware PWR pin toggle
    digitalWrite(MODEM_PWR, LOW);
    delay(1500);
    digitalWrite(MODEM_PWR, HIGH);
    delay(8000);  // Extended boot time
    
    // Verify with multiple AT commands
    int atSuccess = 0;
    for (int i = 0; i < 5; i++) {
        SerialAT.println("AT");
        delay(200);
        
        if (SerialAT.find("OK")) {
            atSuccess++;
        }
        delay(100);
    }
    
    if (atSuccess >= 3) {
        SerialMon.println("NetworkTask: Modem powered on successfully");
        return true;
    }
    
    // Method 2: Try software reset if hardware method didn't work well
    SerialMon.println("NetworkTask: Hardware power cycle uncertain, trying software reset...");
    if (xSemaphoreTake(modemMutex, pdMS_TO_TICKS(5000)) == pdTRUE) {
        bool restarted = modem.restart();
        xSemaphoreGive(modemMutex);
        
        if (restarted) {
            SerialMon.println("NetworkTask: Modem software restart successful");
            return true;
        }
    }
    
    return false;
}

bool initializeModem() {
    sendLedCommand(LED_GSM_INIT);
    SerialMon.println("NetworkTask: Initializing modem...");
    
    if (!modem.restart()) {
        SerialMon.println("NetworkTask: Modem restart failed");
        return false;
    }
    
    String modemInfo = modem.getModemInfo();
    SerialMon.print("NetworkTask: Modem Info: ");
    SerialMon.println(modemInfo);
    
    // Wait for network registration
    SerialMon.println("NetworkTask: Waiting for network...");
    
    if (!modem.waitForNetwork(180000)) {  // 3 minute timeout
        SerialMon.println("NetworkTask: Network registration timeout");
        return false;
    }
    
    SerialMon.println("NetworkTask: Network registered");
    return true;
}

bool connectToMQTT() {
    sendLedCommand(LED_MQTT_CONNECTING);
    SerialMon.println("NetworkTask: Connecting to MQTT...");
    
    if (xSemaphoreTake(mqttMutex, pdMS_TO_TICKS(5000)) != pdTRUE) {
        return false;
    }
    
    // Generate unique client ID with random suffix
    String clientId = "GMS-" + DEVICE_ID + "-" + String(esp_random() & 0xFFFF, HEX);
    
    // Set last will testament
    String willTopic = String(pubTopic) + "/status";
    String willMessage = "{\"device\":\"" + DEVICE_ID + "\",\"status\":\"offline\"}";
    
    // Try connection with different socket timeouts
    bool connected = false;
    int timeouts[] = {15, 30, 45};
    
    for (int i = 0; i < 3 && !connected; i++) {
        mqtt.setSocketTimeout(timeouts[i]);
        
        connected = mqtt.connect(clientId.c_str(), 
                                mqttUser, 
                                mqttPass,
                                willTopic.c_str(),
                                1,  // QoS 1 for will message
                                1,  // retain
                                willMessage.c_str());
        
        if (!connected) {
            SerialMon.printf("MQTT attempt %d failed: %s\n", 
                           i+1, mqttStateToText(mqtt.state()).c_str());
            delay(1000);
        }
    }
    
    if (connected) {
        SerialMon.println("NetworkTask: MQTT connected");
        
        // Subscribe with QoS 1
        if (mqtt.subscribe(subTopic, 1)) {
            SerialMon.printf("NetworkTask: Subscribed to %s (QoS 1)\n", subTopic);
        }
        
        // Publish connection event
        char onlineMsg[100];
        snprintf(onlineMsg, sizeof(onlineMsg),
                "{\"device\":\"%s\",\"event\":\"connected\",\"rssi\":%d,\"fw\":\"%s\"}",
                DEVICE_ID.c_str(), modem.getSignalQuality(), FW_VERSION);
        mqtt.publish(pubTopic, onlineMsg, 1);  // QoS 1
        
    } else {
        SerialMon.printf("NetworkTask: MQTT connection failed, rc=%d (%s)\n", 
                        mqtt.state(), mqttStateToText(mqtt.state()).c_str());
    }
    
    xSemaphoreGive(mqttMutex);
    return connected;
}





void mqttCallback(char* topic, byte* payload, unsigned int length) {
    String message;
    for (unsigned int i = 0; i < length; i++) {
        message += (char)payload[i];
    }
    message.trim();
    
    SerialMon.printf("NetworkTask: MQTT message - Topic: %s, Payload: %s\n", topic, message.c_str());
    sendLedCommand(LED_MQTT_RECEIVE);
    
    if (strcmp(topic, subTopic) == 0) {
        // Parse JSON or command
        if (message.indexOf("ping") != -1) {
            MQTTMessage response;
            snprintf(response.topic, sizeof(response.topic), "%s", pubTopic);
            snprintf(response.payload, sizeof(response.payload), "{\"device\":\"%s\",\"rssi\":\"%d\"}", DEVICE_ID.c_str(), modem.getSignalQuality());
            xQueueSend(mqttPublishQueue, &response, pdMS_TO_TICKS(100));
        }
        else if (message.indexOf("rssi") != -1) {
            int rssi = modem.getSignalQuality();
            MQTTMessage response;
            snprintf(response.topic, sizeof(response.topic), "%s", pubTopic);
            snprintf(response.payload, sizeof(response.payload), 
                     "{\"device\":\"%s\",\"rssi\":%d}", DEVICE_ID.c_str(), rssi);
            xQueueSend(mqttPublishQueue, &response, pdMS_TO_TICKS(100));
        }


        else if (message.indexOf("alarm off") != -1) {
            digitalWrite(HORN_PIN, HIGH);
            vTaskDelay(pdMS_TO_TICKS(300));
            digitalWrite(HORN_PIN, LOW);
            vTaskDelay(pdMS_TO_TICKS(200));
            digitalWrite(HORN_PIN, HIGH);
            vTaskDelay(pdMS_TO_TICKS(300));
            digitalWrite(HORN_PIN, LOW);
        }
        else if (message.indexOf("alarm on") != -1) {
            digitalWrite(HORN_PIN, HIGH);
        }


        else if (message.indexOf("ota") != -1) {
            // Create OTA task
            createOTATask();
        }
        else if (message.indexOf("restart") != -1) {
            MQTTMessage response;
            snprintf(response.topic, sizeof(response.topic), "%s", pubTopic);
            snprintf(response.payload, sizeof(response.payload), 
                     "{\"device\":\"%s\",\"status\":\"restarting\"}", DEVICE_ID.c_str());
            xQueueSend(mqttPublishQueue, &response, pdMS_TO_TICKS(100));
            delay(1000);
            ESP.restart();
        }
    }
}

        tasksSuspended = false;

        vTaskResume(networkTaskHandle);

    }
}

bool performOTAUpdate() {
    SerialMon.println("\n[OTA] Starting OTA update");
    
    // DON'T disconnect GSM here - OTA needs active connection!
    // Keep GPRS connected throughout OTA
    
    // Create a dedicated client for OTA
    TinyGsmClient otaClient(modem);
    
    unsigned long otaStartTime = millis();
    const unsigned long OTA_TOTAL_TIMEOUT = 300000; // 5 minutes total timeout
    
    // ========== STEP 1: CONNECT TO OTA SERVER ==========
    SerialMon.printf("[OTA] Connecting to %s:%d\n", otaHost, otaPort);
    
    int connectAttempts = 0;
    bool serverConnected = false;
    
    while (connectAttempts < 3 && !serverConnected) {
        connectAttempts++;
        SerialMon.printf("[OTA] Connection attempt %d/3\n", connectAttempts);
        
        if (otaClient.connect(otaHost, otaPort)) {
            serverConnected = true;
            SerialMon.println("[OTA] Connected to server");
        } else {
            SerialMon.println("[OTA] Connection failed");
            if (connectAttempts < 3) {
                delay(2000);
                
                // Check if GSM is still connected
                if (!modem.isGprsConnected()) {
                    SerialMon.println("[OTA] GPRS lost during OTA, aborting");
                    return false;
                }
            }
        }
    }
    
    if (!serverConnected) {
        SerialMon.println("[OTA] Failed to connect to server");
        otaClient.stop();
        return false;
    }
    
    // ========== STEP 2: SEND HTTP REQUEST ==========
    SerialMon.printf("[OTA] Requesting firmware: %s\n", firmwarePath);
    
    String request = String("GET ") + firmwarePath + " HTTP/1.1\r\n" +
                    "Host: " + otaHost + "\r\n" +
                    "User-Agent: ESP32-OTA\r\n" +
                    "Connection: keep-alive\r\n" +
                    "\r\n";
    
    otaClient.print(request);
    
    // ========== STEP 3: PARSE HTTP RESPONSE ==========
    unsigned long headerTimeout = millis();
    const unsigned long HEADER_TIMEOUT = 10000;
    bool headersComplete = false;
    int contentLength = 0;
    bool httpOK = false;
    
    SerialMon.println("[OTA] Waiting for HTTP response...");
    
    while (millis() - headerTimeout < HEADER_TIMEOUT && !headersComplete) {
        if (otaClient.available()) {
            String line = otaClient.readStringUntil('\n');
            line.trim();
            
            SerialMon.println("[OTA HDR] " + line);
            
            if (line.length() == 0) {
                // Empty line indicates end of headers
                headersComplete = true;
                break;
            }
            
            if (line.startsWith("HTTP/1.")) {
                if (line.indexOf("200") > 0) {
                    httpOK = true;
                }
            }
            
            if (line.startsWith("Content-Length:")) {
                contentLength = line.substring(15).toInt();
            }
        }
        vTaskDelay(pdMS_TO_TICKS(10));
    }
    
    if (!headersComplete) {
        SerialMon.println("[OTA] HTTP header timeout");
        otaClient.stop();
        return false;
    }
    
    if (!httpOK) {
        SerialMon.println("[OTA] HTTP not OK");
        otaClient.stop();
        return false;
    }
    
    if (contentLength <= 0) {
        SerialMon.println("[OTA] Invalid content length");
        otaClient.stop();
        return false;
    }
    
    SerialMon.printf("[OTA] Firmware size: %d bytes\n", contentLength);
    
    // ========== STEP 4: PREPARE UPDATE ==========
    if (!Update.begin(contentLength, U_FLASH)) {
        SerialMon.printf("[OTA] Update.begin failed: %s\n", Update.errorString());
        otaClient.stop();
        return false;
    }
    
    // ========== STEP 5: DOWNLOAD FIRMWARE ==========
    uint8_t buffer[512];
    size_t totalWritten = 0;
    unsigned long lastDataTime = millis();
    unsigned long lastProgressTime = millis();
    const unsigned long DATA_TIMEOUT = 60000; // 30 seconds timeout between data
    const unsigned long PROGRESS_INTERVAL = 1000; // Update progress every second
    
    SerialMon.println("[OTA] Starting firmware download...");
    
    while (totalWritten < contentLength) {
        // Check total timeout
        if (millis() - otaStartTime > OTA_TOTAL_TIMEOUT) {
            SerialMon.printf("[OTA] Total OTA timeout (%lu ms)\n", millis() - otaStartTime);
            Update.abort();
            otaClient.stop();
            return false;
        }
        
        // Check if GSM is still connected
        if (!modem.isGprsConnected()) {
            SerialMon.println("[OTA] GPRS connection lost during download");
            Update.abort();
            otaClient.stop();
            return false;
        }
        
        // Read available data
        if (otaClient.available()) {
            lastDataTime = millis();
            
            int bytesAvailable = otaClient.available();
            int toRead = min(min(bytesAvailable, (int)sizeof(buffer)), 
                           (int)(contentLength - totalWritten));
            
            if (toRead > 0) {
                int bytesRead = otaClient.read(buffer, toRead);
                
                if (bytesRead > 0) {
                    // Write to flash
                    if (Update.write(buffer, bytesRead) != bytesRead) {
                        SerialMon.printf("[OTA] Write error at byte %d\n", totalWritten);
                        Update.abort();
                        otaClient.stop();
                        return false;
                    }
                    
                    totalWritten += bytesRead;
                    
                    // Show progress
                    if (millis() - lastProgressTime > PROGRESS_INTERVAL) {
                        float percent = (100.0 * totalWritten) / contentLength;
                        SerialMon.printf("[OTA] Progress: %.1f%% (%d/%d bytes)\n", 
                                       percent, totalWritten, contentLength);
                        lastProgressTime = millis();
                    }
                }
            }
        } else {
            // No data available, small delay
            vTaskDelay(pdMS_TO_TICKS(100));
            
            // Check for data timeout
            if (millis() - lastDataTime > DATA_TIMEOUT) {
                SerialMon.printf("[OTA] Data timeout after %lu ms (at %d/%d bytes)\n", 
                               millis() - lastDataTime, totalWritten, contentLength);
                Update.abort();
                otaClient.stop();
                return false;
            }
        }
        
        // Feed watchdog
        esp_task_wdt_reset();
    }
    
    // ========== STEP 6: FINALIZE UPDATE ==========
    SerialMon.println("\n[OTA] Download complete, finalizing...");
    
    if (!Update.end()) {
        SerialMon.printf("[OTA] Update.end failed: %s\n", Update.errorString());
        otaClient.stop();
        return false;
    }
    
    if (!Update.isFinished()) {
        SerialMon.println("[OTA] Update not finished");
        otaClient.stop();
        return false;
    }
    
    SerialMon.println("[OTA] Firmware update successful!");
    otaClient.stop();
    
    // Wait a moment and restart
    SerialMon.println("[OTA] Restarting in 3 seconds...");
    delay(3000);
    
    ESP.restart();
    
    return true; // Never reaches here
}
